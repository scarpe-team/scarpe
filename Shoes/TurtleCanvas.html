def forward(len = 100)
  is_step
  x = len * sin(@heading) + @x
  y = len * cos(@heading) + @y
  if @pendown
    line(@x, @y, x, y)
  end
  update_position(x, y)
end

def reset
  clear_internal
  @pendown = true
  @heading = 180 * DEG
  @turtle_angle = 180
  @bg_color = white
  @fg_color = black
  @pen_size = 1
  background_internal @bg_color
  stroke @fg_color
  strokewidth @pen_size
  update_position(@width / 2, @height / 2)
  update_turtle_heading
end

def toggle_pause
  @paused = !@paused
  if !@paused
    @speed = SPEED if @speed.nil?
    step
  end
  @paused
end
